extends SceneTree

const BATTLE := preload("res://scenes/FleetBattle3D.tscn")
const MAP := "res://scenes/WorldMap.tscn"
const HELD := MAP + ".p02-held"
const SEED := 71757
var fails := 0
var checks := 0
var transitions := 0

func _notice(battle: Node) -> Array:
	return battle.find_children("WorldReturnNotice", "Label", true, false).filter(func(child): return child.is_visible_in_tree())

func _facts(o: BattleOutcome) -> Array:
	return [o.kind, o.win, o.limited, o.reason, o.grade, o.cao_status, o.liu_status, o.t_s, o.loss_pct, o.ending, o.line, o.fingerprint, o.seed, o.mode, o.difficulty]

func _capture(name: String) -> void:
	await RenderingServer.frame_post_draw
	var image := root.get_texture().get_image()
	_check("capture-" + name, image != null and image.get_size() == Vector2i(1600, 900) and image.save_png(OS.get_environment("P02_NOTICE_CAPTURE") + "/" + name + ".png") == OK)


func _initialize() -> void:
	call_deferred("_run")

func _check(id: String, ok: bool) -> void:
	checks += 1
	fails += 0 if ok else 1
	print("P02_RETURN_ASSERT ", id, " ", "PASS" if ok else "FAIL")

func _finish() -> void:
	print("P02_RETURN_DONE ", checks, " ", fails)
	quit(0 if fails == 0 else 1)

func _run() -> void:
	var args := OS.get_cmdline_user_args()
	var lane := args[0] if args.size() == 2 else ""
	var variant := args[1] if args.size() == 2 else ""
	var link := root.get_node("WorldLink")
	var map_path := ProjectSettings.globalize_path(MAP).replace("\\", "/")
	var held_path := ProjectSettings.globalize_path(HELD)
	var project_path := ProjectSettings.globalize_path("res://").replace("\\", "/").trim_suffix("/")
	var user_path := OS.get_user_data_dir().replace("\\", "/")
	_check("scope", project_path == OS.get_environment("P02_PC_EXPECT_PROJECT") and user_path == OS.get_environment("P02_PC_EXPECT_USER") and map_path == project_path + "/scenes/WorldMap.tscn")
	_check("arguments", lane in ["result", "pause", "prepare", "result-ok", "pause-ok", "prepare-ok", "standalone-pause", "standalone-result", "standalone-prepare"] and variant in ["before", "after"])
	if fails:
		_finish()
		return
	var phase := lane.trim_suffix("-ok").trim_prefix("standalone-")
	var standalone := lane.begins_with("standalone-")
	var reject := not standalone and not lane.ends_with("-ok")
	var native := not OS.get_environment("P02_NOTICE_CAPTURE").is_empty()
	var brief := BattleBrief.new()
	brief.seed = SEED
	brief.difficulty = "표준"
	brief.mode = "direct"
	var previous := BattleOutcome.new()
	previous.seed = -117
	link.brief = null if standalone else brief
	link.outcome = previous
	link.last_mode = brief.mode
	link.last_difficulty = brief.difficulty
	var battle := BATTLE.instantiate()
	if phase == "prepare":
		battle.profile_patch = func(profile: Dictionary): profile.clear()
	root.add_child(battle)
	current_scene = battle
	var deck = null if phase == "prepare" else battle.presentation.hud
	var button: Button
	if phase == "prepare":
		button = battle.find_child("ExitBattlePrepareError", true, false) as Button
	elif phase == "result":
		deck.begin_battle()
		battle.sim._end(true, "annihilation")
		battle.end_game(true, "P02 fixture")
		await create_timer(2.2).timeout
		button = deck.screens.result.title_btn
	else:
		deck.begin_battle()
		battle._toggle_menu()
		deck.show_screen("pause")
		button = deck.screens.pause.title_btn
	var sim = battle.sim
	var screen: String = deck.screen if deck != null else ""
	var had_state: bool = battle.G.has("state")
	var state: Variant = battle.G["state"] if had_state else null
	var expected_fingerprint: String = sim.fingerprint() if sim != null else ""
	var clock_before: float = sim.st.clock_s() if sim != null else 0.0
	var selected_before: Array = battle.selected.duplicate()
	var result_before: Dictionary = sim.st.result.duplicate(true) if sim != null else {}
	var paused_before := paused
	var guidance: Label = button.get_parent().get_child(2) if phase == "prepare" and button != null else null
	var reason: Label = battle.find_child("Reason", true, false) if phase == "prepare" else null
	var guidance_before := guidance.text if guidance != null else ""
	var reason_before := reason.text if reason != null else ""
	var expected_time: float = sim.st.clock_s() if phase == "result" else 0.0
	_check("battle", current_scene == battle and link.brief == (null if standalone else brief) and link.outcome == previous and link.routed() != standalone)
	_check("phase", (sim == null if phase == "prepare" else sim != null and sim.st.over == (phase == "result")) and (phase == "prepare" or screen == phase))
	_check("state-presence", had_state == (phase != "prepare"))
	_check("state-value", battle.G.is_empty() if phase == "prepare" else had_state and is_same(state, "end" if phase == "result" else "pause"))
	_check("button", button != null and button.is_visible_in_tree() and not button.disabled and button.text == (("종료" if phase == "prepare" else "타이틀로") if standalone else "천하로"))
	_check("map-uncached", not ResourceLoader.has_cached(MAP) and not FileAccess.file_exists(HELD) and FileAccess.file_exists(MAP))
	if fails:
		_finish()
		return
	await process_frame
	var button_rect := button.get_global_rect()
	var focus := root.gui_get_focus_owner()
	_check("hidden", _notice(battle).is_empty())
	if native:
		await create_timer(0.35, true, false, true).timeout
		print("P02_NOTICE_RECT button ", button_rect)
		await _capture("initial")
	if standalone:
		_check("standalone-hidden", _notice(battle).is_empty())
		if phase == "prepare":
			if fails:
				_finish()
				return
			print("P02_RETURN_DONE ", checks, " ", fails)
			button.pressed.emit()
			await create_timer(2.0).timeout
			_check("standalone-quit-failed", false)
			_finish()
			return
		button.pressed.emit()
		await process_frame
		_check("standalone-title", current_scene == battle and deck.screen == "title" and battle.sim != sim and link.p02_notice_requests.is_empty() and link.outcome == previous and _notice(battle).is_empty())
		_finish()
		return
	scene_changed.connect(func(): transitions += 1)
	if reject:
		_check("hide-map", DirAccess.rename_absolute(map_path, held_path) == OK and not FileAccess.file_exists(MAP) and FileAccess.file_exists(HELD) and not ResourceLoader.exists(MAP))
		if not fails:
			var repetitions := 2
			for i in repetitions:
				var tag := "reject-%d-" % (i + 1)
				print("P02_RETURN_ACTION ", tag)
				button.pressed.emit()
				await process_frame
				_check(tag + "scene", current_scene == battle and transitions == 0 and battle.sim == sim)
				_check(tag + "state-presence", battle.G.has("state") == had_state)
				_check(tag + "state-value", battle.G.has("state") and is_same(battle.G["state"], state) if had_state else battle.G.is_empty())
				_check(tag + "ui", button.is_visible_in_tree() and not button.disabled and button.text == "천하로" and (deck == null or deck.screen == screen))
				_check(tag + "settings", link.last_difficulty == "표준" and link.last_mode == "direct")
				_check(tag + "brief", link.brief == brief and link.routed())
				_check(tag + "queue", link.outcome == previous and previous.seed == -117)
				_check(tag + "sim", battle.selected == selected_before and paused == paused_before and (sim == null or (sim.fingerprint() == expected_fingerprint and sim.st.clock_s() == clock_before and sim.st.result == result_before)))
				_check(tag + "focus-geometry", button.get_global_rect() == button_rect and root.gui_get_focus_owner() == focus)
				_check(tag + "guidance", phase != "prepare" or (reason != null and guidance != null and reason.text == reason_before and guidance.text == guidance_before))
				_check(tag + "request", link.p02_notice_requests.size() == i + 1 and ((link.p02_notice_requests[i] is BattleOutcome) if phase == "result" else link.p02_notice_requests[i] == null))
				_check(tag + "actual-error", link.p02_notice_errors.size() == i + 1 and link.p02_notice_errors[i] == 19)
				var visible := _notice(battle)
				_check(tag + "notice", visible.size() == (1 if variant == "after" else 0))
				if variant == "after":
					var label: Label = visible[0] if visible.size() == 1 else null
					_check(tag + "layout", label != null and label.text == "천하로 돌아가지 못했습니다. 다시 눌러 주세요." and label.mouse_filter == Control.MOUSE_FILTER_IGNORE and not label.get_global_rect().intersects(button_rect) and root.get_visible_rect().encloses(label.get_global_rect()) and label.get_minimum_size().y <= label.size.y)
				if native:
					if visible.size() == 1:
						print("P02_NOTICE_RECT warning ", visible[0].get_global_rect())
					await _capture("reject-%d" % (i + 1))
		_check("restore-map", DirAccess.rename_absolute(held_path, map_path) == OK and FileAccess.file_exists(MAP) and not FileAccess.file_exists(HELD) and ResourceLoader.exists(MAP))
		if fails:
			_finish()
			return
	if reject and deck != null and variant == "after":
		deck.show_screen("settings_pause" if phase == "pause" else "title")
		deck.show_screen(phase)
		_check("reentry-hidden", _notice(battle).is_empty())
	print("P02_RETURN_ACTION accept")
	button.pressed.emit()
	for i in range(12):
		await process_frame
	var world := current_scene as WorldMapScreen
	var requests: Array = link.p02_notice_requests
	var errors: Array = link.p02_notice_errors
	_check("accept-request", requests.size() == (3 if reject else 1) and errors.size() == requests.size() and errors.back() == OK)
	_check("accept-old-scene", not is_instance_valid(battle))
	_check("accept-scene", world != null and transitions == 1)
	_check("accept-queue", not link.routed() and link.outcome == null and link.take_outcome() == null)
	_check("accept-settings", link.last_difficulty == "표준" and link.last_mode == "direct")
	if world == null:
		_finish()
		return
	if phase == "result":
		var o := world.outcome
		_check("accept-view", world.state == "result" and o != null and world._result.visible)
		_check("accept-dto", o != null and o == requests.back() and (not reject or (requests.size() == 3 and requests[0] is BattleOutcome and requests[1] is BattleOutcome and requests[0] != requests[1] and requests[0] != o and requests[1] != o and _facts(requests[0]) == _facts(o) and _facts(requests[1]) == _facts(o))))
		_check("accept-facts", o != null and o != previous and o.seed == SEED and o.difficulty == "표준" and o.mode == "direct" and o.win and o.reason == "annihilation" and o.t_s == expected_time and o.fingerprint == expected_fingerprint)
	else:
		_check("accept-null", requests.all(func(request): return request == null))
		_check("accept-view", world.state == "browse" and world.outcome == null and world._cta.visible)
	_finish()
