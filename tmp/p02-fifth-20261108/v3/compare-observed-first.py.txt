def read(path):
    return json.loads(path.read_bytes())

def sha(path):
    with path.open("rb") as f:
        return hashlib.file_digest(f, "sha256").hexdigest()

def same_file(path, record):
    return path.stat().st_size == record["bytes"] and sha(path) == record["sha256"]

def one_match(path, regex):
    matches = re.findall(regex, path.read_text(encoding="utf8"), re.M)
    assert len(matches) == 1, (path, regex, matches)
    return matches[0]

def run_report(stage, names):
    rows = read(stage / "processes.json")
    assert [r["name"] for r in rows] == list(names)
    for name, row in zip(names, rows):
        dirname = stage / name
        assert row["exit"] == (1 if name == CLEAN else 0) and row["marker"] and not row["script_error"] and not row["timeout"], name
        assert row == read(dirname / "process.json")
        assert row["user_start"] == read(dirname / "user-start.json")
        assert Path(row["user_probe"]["dir"]).resolve() == Path(row["expected_user"]).resolve()
        expected_settings = name == RETURNING
        assert set(row["user_start"]) == ({"settings.cfg"} if expected_settings else set())
        assert row["user_probe"]["initial_files"] == (["settings.cfg"] if expected_settings else [])
        if expected_settings:
            assert row["user_start"]["settings.cfg"] == row["user_end"]["settings.cfg"]
            assert row["user_probe"]["settings_sha"] == row["user_start"]["settings.cfg"]["sha256"]
        for stream in ("stdout", "stderr"):
            assert sha(dirname / stream) == row[stream]
        if "result_sha256" in row:
            data = dirname / ("observations.json" if name == HEADLESS else "result.json")
            assert sha(data) == row["result_sha256"]
        if name in (COMMON, BOUNDARY):
            actual = read(dirname / "result.json")
            assert actual["failures"] == 0
            count, prefix = (18, "A") if name == COMMON else (7, "B")
            assert sorted(actual["rows"]) == [prefix + f"{i:02d}" for i in range(1, count + 1)]
        user = Path(row["expected_user"])
        assert all(same_file(user / p, info) for p, info in row["user_end"].items()), name
    return dict(zip(names, rows))

    value = (stage / CLEAN / "stderr").read_text(encoding="utf8")
