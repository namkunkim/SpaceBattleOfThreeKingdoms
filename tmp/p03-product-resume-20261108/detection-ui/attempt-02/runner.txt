extends SceneTree

func _initialize() -> void:
	call_deferred("_run")

func check(ok: bool, label: String) -> bool:
	print("CHECK ", ok, " ", label)
	if not ok:
		quit(1)
	return ok

func frame(battle: Node, path: String) -> bool:
	battle.presentation._process(0.0)
	battle.presentation.hud.overlay.queue_redraw()
	await RenderingServer.frame_post_draw
	var image := root.get_texture().get_image()
	return check(image != null and not image.is_empty() and image.save_png(path) == OK, path)

func _run() -> void:
	var battle := (load("res://scenes/FleetBattle3D.tscn") as PackedScene).instantiate()
	root.add_child(battle)
	battle.set_process(false)
	battle.presentation.hud.guard.enabled = false
	battle.presentation.hud.begin_battle()
	var sim: BattleSim = battle.sim
	var detection: Detection = sim.detect
	var targets: Array[FleetState] = [null, null]
	for f in sim.st.fleets:
		if not f.dead and f.max_hull > 0 and targets[f.side] == null:
			targets[f.side] = f
	if not check(targets[0] != null and targets[1] != null, "targets"): return
	var center: Vector2 = battle.cam_pos
	for f in sim.st.fleets:
		if not targets.has(f):
			f.dead = true
	targets[0].pos = center + Vector2(-45, 0)
	targets[1].pos = center + Vector2(45, 0)
	sim.st.tick = 1
	detection.step()
	for side in 2:
		if not check(detection.state(side, targets[1-side].id) == "confirmed", "confirmed side %d" % side): return
	var positions: Array[Vector2] = [targets[0].pos, targets[1].pos]
	targets[0].pos = center + Vector2(-550, 0)
	targets[1].pos = center + Vector2(550, 0)
	detection.D = detection.D.duplicate(true)
	detection.D.estimated = 1000000
	detection.D.confirmed = 1000001
	detection.D.view_r = 1
	detection.D.radar_r = 1
	var memory := BattleRules.ticks(float(detection.D.memory_s), sim.st.hz)
	var expiry := memory + BattleRules.ticks(float(detection.D.lost_s), sim.st.hz)
	var period: int = detection._period
	sim.st.tick = 1 + (floori(float(memory) / period) + 1) * period
	detection.step()
	for side in [battle.detection_test_side]:
		battle._pump()
		var target := targets[1-side]
		var v = battle.vm.fleet(target.id)
		if not check(v != null and v.contact == "lost" and v.pos == positions[1-side], "lost at last known position side %d" % side): return
		print("SIDE ", side, " LOST_TICK ", sim.st.tick, " OLD ", v.pos, " HIDDEN ", target.pos, " HIDDEN_SCREEN ", battle.w2s(target.pos))
		if not await frame(battle, "res://lost-%d.png" % side): return
	sim.st.tick = 1 + (floori(float(expiry) / period) + 1) * period
	detection.step()
	print("EXPIRY_TICK ", sim.st.tick, " EVENTS ", sim.get("_events"))
	for side in [battle.detection_test_side]:
		battle._pump()
		var target := targets[1-side]
		if not check(detection.state(side, target.id) == "" and battle.vm.fleet(target.id) == null, "contact removed from view side %d" % side): return
		if not await frame(battle, "res://expired-%d.png" % side): return
		if not check(not battle.presentation.renderer.vis.has(target.id), "renderer ID removed side %d" % side): return
	print("DETECTION_UI_PASS")
	for key in ["detect", "salvo", "supply", "morale", "victory", "cai", "decisions", "chain", "cmd", "assault", "skills"]:
		var module = sim.get(key)
		if module != null:
			module.set("sim", null)
	battle.queue_free()
	await process_frame
	quit(0)
