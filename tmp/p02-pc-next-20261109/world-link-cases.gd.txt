func _pc_check(id: String, ok: bool) -> void:
	print("PC_DIRECT_ROW ", id, " ", "PASS" if ok else "FAIL")
	_check(ok, id)


func _pc_find_button(parent: Node, label: String) -> Button:
	var buttons: Array = []
	_buttons(parent, buttons)
	for button in buttons:
		if button.text == label:
			return button
	return null


func _pc_link_matches(link: Node, prior: Array) -> bool:
	return link.brief == prior[0] and link.outcome == prior[1] and link.last_difficulty == prior[2] and link.last_mode == prior[3]


func _pc_seed_link(link: Node, sentinel: BattleOutcome) -> Array:
	var old := BattleBrief.new()
	old.seed = 991
	link.brief = old
	link.outcome = sentinel
	link.last_difficulty = "쉬움"
	link.last_mode = "auto"
	return [old, sentinel, "쉬움", "auto"]


func _pc_result_labels(s: WorldMapScreen) -> Array:
	return [s._res_labels.title.text, s._res_labels.seed.text, s._res_labels.ending.text, s._res_labels.loss_a.text]


func _pc_reject() -> void:
	var battle_file := "res://scenes/FleetBattle3D.tscn"
	_pc_check("F1", not FileAccess.file_exists(battle_file))
	_pc_check("F2", not ResourceLoader.has_cached(battle_file))
	_pc_check("F3", _pc_paths_match())
	if fails != 0:
		return
	var link := root.get_node("WorldLink")
	var s := _screen_new()
	await _frames(4)
	var sentinel := BattleOutcome.new()
	var prior := _pc_seed_link(link, sentinel)
	var requested := BattleBrief.new()
	requested.seed = 777
	requested.mode = "direct"
	requested.difficulty = "표준"
	var code: Variant = link.call("go_battle", requested)
	_pc_check("R1", typeof(code) == TYPE_INT and code == ERR_CANT_OPEN)
	_pc_check("R2", current_scene == s and s.is_inside_tree())
	_pc_check("R3", link.brief == prior[0])
	_pc_check("R4", link.outcome == prior[1])
	_pc_check("R5", link.last_difficulty == prior[2])
	_pc_check("R6", link.last_mode == prior[3])
	prior = _pc_seed_link(link, sentinel)
	var local := BattleBrief.new()
	s.brief = local
	s.open_card()
	await _seconds(1.5)
	s._set_difficulty("표준")
	s._set_mode("direct")
	var button := _pc_find_button(s._card, "출  격")
	if button == null:
		_check(false, "A fixture sortie button missing")
	else:
		button.emit_signal("pressed")
		s._skip.emit_signal("pressed")
		_pc_check("A1", current_scene == s and s.is_inside_tree())
		_pc_check("A2", _pc_link_matches(link, prior))
		_pc_check("A3", s.brief == local)
		_pc_check("A4", s.state == "browse")
		_pc_check("A5", not s.canvas.input_lock)
		_pc_check("A6", not s._caption.visible and not s._skip.visible)
		_pc_check("A7", s._card.visible and button.is_visible_in_tree() and not button.disabled)
		_pc_check("A8", s._chip.visible and not s._chip_label.text.is_empty())
		_pc_check("A9", s.resolver == null and not s._resolve.visible)
		s._skip.emit_signal("pressed")
		await _frames(4)
		_pc_check("A10", s.state == "browse" and current_scene == s and _pc_link_matches(link, prior))
	s.queue_free()
	await _frames(3)
	await _pc_reject_result(link, "BS", "같은 판 다시")
	await _pc_reject_result(link, "BN", "새 판")


func _pc_reject_result(link: Node, prefix: String, label: String) -> void:
	var result := BattleOutcome.new()
	result.kind = "alliance_win"
	result.win = true
	result.cao_status = "unhurt"
	result.seed = 777
	result.mode = "direct"
	result.difficulty = "표준"
	link.outcome = result
	var s := _screen_new()
	await _frames(4)
	await _seconds(0.75)
	var sentinel := BattleOutcome.new()
	var prior := _pc_seed_link(link, sentinel)
	var local := BattleBrief.new()
	s.brief = local
	s._set_difficulty("표준")
	s._set_mode("direct")
	var labels := _pc_result_labels(s)
	var owners: Dictionary = s.canvas.owners_rgn.duplicate()
	var button := _pc_find_button(s._result, label)
	if s.outcome != result or button == null:
		_check(false, prefix + " fixture result missing")
		s.queue_free()
		await _frames(3)
		return
	var expected := 777
	if prefix == "BN":
		seed(4421)
		expected = randi() % 1000000
		seed(4421)
	button.emit_signal("pressed")
	_pc_check(prefix + "1", s.state == "sortie" and s.brief.seed == expected and s.brief.mode == "direct")
	UiSound.history.clear()
	s._skip.emit_signal("pressed")
	_pc_check(prefix + "2", current_scene == s and s.is_inside_tree())
	_pc_check(prefix + "3", _pc_link_matches(link, prior))
	_pc_check(prefix + "4", s.brief == local)
	_pc_check(prefix + "5", s.outcome == result)
	_pc_check(prefix + "6", s.state == "result" and s._result.visible)
	_pc_check(prefix + "7", _pc_result_labels(s) == labels)
	var same := _pc_find_button(s._result, "같은 판 다시")
	var again := _pc_find_button(s._result, "새 판")
	_pc_check(prefix + "8", same != null and again != null and not same.disabled and not again.disabled)
	_pc_check(prefix + "9", not s.canvas.input_lock and not s._caption.visible and not s._skip.visible)
	_pc_check(prefix + "10", s._chip.visible and not s._chip_label.text.is_empty())
	_pc_check(prefix + "11", link.outcome == sentinel and s.resolver == null)
	_pc_check(prefix + "12", not UiSound.history.has("decision"))
	s._skip.emit_signal("pressed")
	await _seconds(0.7)
	_pc_check(prefix + "13", current_scene == s and s.state == "result" and s.outcome == result and link.outcome == sentinel and s.canvas.owners_rgn == owners and not UiSound.history.has("decision"))
	s.queue_free()
	await _frames(3)


func _pc_paths_match() -> bool:
	var project := ProjectSettings.globalize_path("res://").simplify_path().replace("\\", "/").trim_suffix("/")
	var user := OS.get_user_data_dir().simplify_path().replace("\\", "/").trim_suffix("/")
	print("PC_DIRECT_PATHS ", JSON.stringify({"project": project, "user": user}))
	return project == OS.get_environment("P02_PC_EXPECT_PROJECT") and user == OS.get_environment("P02_PC_EXPECT_USER")
