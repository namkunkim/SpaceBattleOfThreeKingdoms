# PC0 reader. Counters belong to this trace and never control the host.
var pc0_phase := ""
var pc0_epoch := 0
var pc0_start_ms := 0
var pc0_second := false
var pc0_advance := {}
var pc0_terminal_end := false

func pc0_reset() -> void:
	pc0_phase = ""
	pc0_epoch = 0
	pc0_second = false
	pc0_advance.clear()
	pc0_terminal_end = false

func pc0_sample(host: Node, ui: Dictionary) -> void:
	var s: BattleState = host.sim.st
	var phase := ""
	if s.over:
		phase = "terminal"
	elif host.placing and ui.state == "play":
		phase = "placing"
	elif not host.placing and ui.state == "pause":
		phase = "paused"
	elif not host.placing and ui.state == "play":
		phase = "running"
	var now := Time.get_ticks_msec()
	var data := ui.duplicate()
	var screen: String = host.presentation.hud.screen if host.presentation and host.presentation.hud else ""
	data.merge({"phase": phase, "epoch": pc0_epoch, "host_id": str(host.get_instance_id()),
		"sim_id": str(host.sim.get_instance_id()), "screen": screen,
		"log_size": host.sim.command_log.size(), "sample_ms": now})
	if phase != pc0_phase:
		write(host, "pc0_phase_exit", {"phase": pc0_phase, "epoch": pc0_epoch,
			"two_samples": pc0_second})
		pc0_advance.clear()
		if phase == "running" and pc0_phase in ["placing", "paused"] and pc0_second:
			pc0_advance = {"from": pc0_phase, "from_epoch": pc0_epoch}
		pc0_phase = phase
		pc0_epoch += 1
		data.epoch = pc0_epoch
		pc0_start_ms = now
		pc0_second = false
		if phase in ["placing", "paused", "terminal"]:
			data.point = 1
			if phase == "placing":
				data.roster = []
				for f in s.fleets:
					if f.side == 0:
						data.roster.append({"side": 0, "id": f.id, "sq": f.sq_id, "name": f.name,
							"faction": f.faction, "is_flag": f.is_flag, "control": f.control})
			write(host, "pc0_phase", data)
	elif phase in ["placing", "paused", "terminal"] and not pc0_second and now - pc0_start_ms >= 2000:
		pc0_second = true
		data.point = 2
		write(host, "pc0_phase", data)
	if phase == "running" and not pc0_advance.is_empty():
		if s.tick > pc0_last_frozen_tick and s.clock_ms > pc0_last_frozen_clock:
			data.merge(pc0_advance)
			write(host, "pc0_advance", data)
			pc0_advance.clear()
	if phase in ["placing", "paused", "terminal"]:
		pc0_last_frozen_tick = s.tick
		pc0_last_frozen_clock = s.clock_ms
	if phase == "terminal" and ui.state == "end" and screen == "result" and not pc0_terminal_end:
		pc0_terminal_end = true
		write(host, "pc0_end_screen", data)

var pc0_last_frozen_tick := -1
var pc0_last_frozen_clock := -1

func pc0_command(host: Node, data: Dictionary) -> Dictionary:
	var copy := data.duplicate()
	var log: Array[Dictionary] = host.sim.command_log
	var ids := []
	for f in host.selected:
		ids.append(f.id)
	copy.pc0 = {"log_size": log.size(), "last": log.back() if not log.is_empty() else null,
		"selected": ids, "host_id": str(host.get_instance_id()), "sim_id": str(host.sim.get_instance_id()),
		"focus": host.get_window().has_focus(), "field": host.field(), "edge": host.sim.R.edge}
	return copy
