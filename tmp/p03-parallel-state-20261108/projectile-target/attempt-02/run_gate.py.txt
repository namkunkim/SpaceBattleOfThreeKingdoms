import argparse
import hashlib
import json
import os
import re
import subprocess
import sys
from pathlib import Path

from windows_job import CREATE_SUSPENDED, LauncherJob

HERE = Path(__file__).resolve().parent
ROOT = HERE.parents[2]
ENGINE = Path("C:/Tools/Godot/Godot_v4.7.2-stable_win64_console.exe")
BODY = ENGINE.with_name("Godot_v4.7.2-stable_win64.exe")
PLAN = HERE.parent / "PLAN-PROJECTILE-TARGET.md"
APPROVED = "9d9925d3ba55c00226d4140801e2b73bf690603bf4470237d884d5482df6928c"
MANIFEST = HERE / "manifest.json"
ENV = dict(os.environ, GIT_OPTIONAL_LOCKS="0", PYTHONDONTWRITEBYTECODE="1")
NAMES = ["launch_and_exact_candidate", "positive_rebind", "atomic_reject_and_advance"]


def digest(data):
    return hashlib.sha256(data).hexdigest()


def record(path, data):
    with path.open("x", encoding="utf-8") as stream:
        json.dump(data, stream, indent=2, ensure_ascii=False)
        stream.write("\n")


def git(*args):
    return subprocess.check_output(["git", "-c", "core.longpaths=true", *args], cwd=ROOT, env=ENV)


def snapshot():
    paths = [
        ENGINE, BODY, Path(sys.executable), Path(os.environ["P03_POWERSHELL"]),
        PLAN, HERE.parent / "REVIEW-PLAN-PROJECTILE-TARGET.md", HERE / "run_gate.py",
        HERE / "windows_job.py", HERE / "projectile_target.gd", HERE / ".gdignore",
        ROOT / "project.godot", ROOT / ".godot/global_script_class_cache.cfg",
        ROOT / "data/bundle-layout.json", ROOT / "data/profiles/poc_corridor.json",
        ROOT / "data/profiles/combat_m3.json",
        *sorted((ROOT / "core/battle").rglob("*.gd")),
    ]
    result = {}
    seen = set()
    for path in paths:
        path = path.resolve(strict=True)
        normalized = str(path).replace("\\", "/")
        if normalized.casefold() in seen:
            raise ValueError(f"duplicate input: {normalized}")
        seen.add(normalized.casefold())
        data = path.read_bytes()
        result[normalized] = {"size": len(data), "sha256": digest(data)}
    if digest(PLAN.read_bytes()) != APPROVED:
        raise ValueError("unapproved plan")
    return {"head": git("rev-parse", "HEAD").decode().strip(), "files": result}


def extract(stdout, prefix):
    matches = [line[len(prefix):] for line in stdout.splitlines() if line.startswith(prefix)]
    return json.loads(matches[0]) if len(matches) == 1 else None


def run_case(parent, name, force):
    out = parent / name
    out.mkdir()
    command = [str(ENGINE), "--headless", "--path", str(ROOT), "--script",
               "res://" + (HERE / "projectile_target.gd").relative_to(ROOT).as_posix()]
    if force:
        command += ["--", "--force-gate-failure"]
    timeout = False
    termination = None
    with LauncherJob() as job, (out / "stdout.bin").open("xb") as stdout_file, (out / "stderr.bin").open("xb") as stderr_file:
        process = subprocess.Popen(command, cwd=ROOT, env=ENV, stdout=stdout_file, stderr=stderr_file,
                                   creationflags=CREATE_SUSPENDED)
        try:
            thread = job.assign_and_resume(process.pid)
        except OSError:
            if job.assigned:
                job.terminate()
            else:
                process.kill()
            process.wait(timeout=5)
            raise
        try:
            process.wait(timeout=60)
        except subprocess.TimeoutExpired:
            timeout = True
            termination = job.terminate()
            process.wait(timeout=5)
        accounting = job.accounting()
    stdout = (out / "stdout.bin").read_bytes()
    stderr = (out / "stderr.bin").read_bytes()
    good = extract(stdout, b"P03_PROJECTILE_TARGET_PASS ")
    bad = extract(stdout, b"P03_PROJECTILE_TARGET_FAIL ")
    payload = bad if force else good
    diagnostics = {
        label: len(re.findall(pattern, stdout + b"\n" + stderr, re.MULTILINE))
        for label, pattern in {
            "script_error": rb"^SCRIPT ERROR:", "error": rb"^ERROR:", "warning": rb"^WARNING:",
            "objectdb": rb"ObjectDB", "resources": rb"resources? still in use",
        }.items()
    }
    identity = bool(payload and payload["engine_pid"] != process.pid
                    and Path(payload["engine_path"]).resolve() == BODY.resolve())
    expected = 1 if force else 0
    fixture = bool(payload and ((force and good is None and payload["checks"] == 1
                   and payload["failures"] == ["intentional_gate_failure"] and payload["cases"] == [])
                   or (not force and bad is None and payload["cases"] == NAMES and payload["failures"] == [])))
    passed = bool(not timeout and process.returncode == expected and fixture and identity
                  and not any(diagnostics.values()) and accounting["active"] == 0 and accounting["total"] >= 2)
    result = {
        "command": command, "expected_exit": expected, "actual_exit": process.returncode,
        "gate_exit": 124 if timeout else process.returncode, "launcher_pid": process.pid, "launcher_thread": thread,
        "timeout": timeout, "termination": termination, "job": accounting, "diagnostics": diagnostics,
        "engine_identity_matches": identity, "success_marker": good, "failure_marker": bad, "passed": passed,
    }
    record(out / "run.json", result)
    return result


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--freeze", action="store_true")
    parser.add_argument("--manifest-sha256")
    parser.add_argument("--attempt", type=int, choices=(1, 2, 3))
    args = parser.parse_args()
    if args.freeze:
        record(MANIFEST, snapshot())
        print(digest(MANIFEST.read_bytes()))
        return 0
    if args.attempt is None or not args.manifest_sha256:
        parser.error("attempt and approved manifest SHA required")
    out = HERE / f"attempt-{args.attempt:02}"
    out.mkdir()
    manifest_data = MANIFEST.read_bytes()
    before = snapshot()
    record(out / "input-before.json", before)
    (out / "git-before.z").write_bytes(git("status", "--porcelain=v1", "-z", "--untracked-files=all"))
    if digest(manifest_data) != args.manifest_sha256 or before != json.loads(manifest_data):
        raise ValueError("manifest SHA or input mismatch")
    main_case = run_case(out, "fixture", False)
    forced = run_case(out, "forced_failure", True)
    after = snapshot()
    record(out / "input-after.json", after)
    (out / "git-after.z").write_bytes(git("status", "--porcelain=v1", "-z", "--untracked-files=all"))
    stable = before == after and digest(MANIFEST.read_bytes()) == args.manifest_sha256
    passed = stable and main_case["passed"] and forced["passed"]
    record(out / "suite.json", {"passed": passed, "input_stable": stable, "inputs": len(before["files"]),
                               "fixture_exit": main_case["actual_exit"], "forced_exit": forced["actual_exit"]})
    if main_case["timeout"] or forced["timeout"]:
        return 124
    return int(not passed)


if __name__ == "__main__":
    try:
        raise SystemExit(main())
    except (OSError, ValueError, KeyError, subprocess.SubprocessError) as error:
        print(f"gate infrastructure error: {error}", file=sys.stderr)
        raise SystemExit(2)
