extends SceneTree

const Private := preload("res://core/battle/operation_private_state.gd")
const NAMES := ["setup_and_exact_input", "positive_replacement", "reject_is_atomic_and_can_advance"]
var failures: Array[String] = []
var checks := 0
var names: Array[String] = []

func _initialize() -> void:
	call_deferred("_run")

func _check(condition: bool, label: String) -> bool:
	checks += 1
	if not condition:
		failures.append(label)
	return condition

func _metadata(value: Variant) -> Array:
	if value is Dictionary:
		return [value.get_typed_key_builtin(), value.get_typed_key_class_name(), value.get_typed_key_script(),
			value.get_typed_value_builtin(), value.get_typed_value_class_name(), value.get_typed_value_script()]
	return [value.get_typed_builtin(), value.get_typed_class_name(), value.get_typed_script()]

func _diff(a: Variant, b: Variant, path: String, diff: Array[String]) -> void:
	if typeof(a) != typeof(b):
		diff.append(path)
	elif a is Dictionary:
		if _metadata(a) != _metadata(b) or not _exact(a.keys(), b.keys()):
			diff.append(path)
		else:
			for key in a:
				_diff(a[key], b[key], path + "." + str(key), diff)
	elif a is Array:
		if _metadata(a) != _metadata(b) or a.size() != b.size():
			diff.append(path)
		else:
			for i in a.size():
				_diff(a[i], b[i], path + "." + str(i), diff)
	elif a is Object:
		if not is_same(a, b):
			diff.append(path)
	elif a != b:
		diff.append(path)

func _exact(a: Variant, b: Variant) -> bool:
	var diff: Array[String] = []
	_diff(a, b, "", diff)
	return diff.is_empty()

func _finish() -> void:
	var record := {"checks": checks, "cases": names, "failures": failures,
		"engine_pid": OS.get_process_id(), "engine_path": OS.get_executable_path()}
	if failures.is_empty() and names.size() == NAMES.size():
		print("P03_REINF_MS_PASS " + JSON.stringify(record))
		quit(0)
	else:
		print("P03_REINF_MS_FAIL " + JSON.stringify(record))
		quit(1)

func _event(kind: String, id: int, pos: Vector2, value: Variant) -> Dictionary:
	return {"tick": 0, "kind": kind, "sq": id, "other": -1, "pos": pos, "value": value}

func _guard(owner: OperationPrivateState) -> bool:
	var objects: Array[Object] = [owner, owner._sim, owner._sim.st, owner._sim.st.fleets[0]]
	var files := ["operation_private_state", "battle_sim", "battle_state", "fleet_state"]
	for i in objects.size():
		if objects[i].get_script().resource_path != "res://core/battle/" + files[i] + ".gd":
			return false
	return true

func _run() -> void:
	if "--force-gate-failure" in OS.get_cmdline_user_args():
		_check(0 == 1, "intentional_gate_failure")
		_finish()
		return
	var profile := {"profile_id": "p03-reinf-ms", "rules": PocSetup.profile().rules.duplicate(true),
		"ally": [{"name": "A", "role": "line", "ships": 10, "lv": 1, "x": 500, "y": 500, "p": 0, "flag": true, "wait": 1000}],
		"foe": [{"name": "B", "role": "line", "ships": 10, "lv": 1, "x": 1500, "y": 1000, "p": 0, "flag": true, "wait": 1000}], "reinf": []}
	var command := {"tick": -1, "side": 0, "ids": [1], "kind": "stop",
		"target_id": -1, "point": Vector2.ZERO, "args": {}}
	var events := [_event("spawn", 1, Vector2.ZERO, null), _event("spawn", 2, Vector2.ZERO, null)]
	var owners: Array[OperationPrivateState] = []
	for label in ["A", "B", "C"]:
		var created := Private.create(profile, 41, 10, [])
		if not _check(created.ok, "create_" + label):
			_finish()
			return
		var owner: OperationPrivateState = created.owner
		owners.append(owner)
		_check(_guard(owner), "class_guard_" + label)
		_check(_exact(BattleSim.command(0, [1], "stop"), command), "command_oracle_" + label)
		_check(_exact(owner.queue(BattleSim.command(0, [1], "stop")), {"ok": true}), "queue_" + label)
	var a := owners[0]
	var b := owners[1]
	var c := owners[2]
	var p_result := b.capture_completed()
	if not _check(p_result.ok, "capture_p"):
		_finish()
		return
	var p: Dictionary = p_result.packet
	var m := p.duplicate(true)
	m.view.state.reinf_ms = 0
	var changed: Array[String] = []
	_diff(p, m, "", changed)
	_check(changed == [".view.state.reinf_ms"], "exact_single_path")
	_check(typeof(m.view.state.reinf_ms) == TYPE_INT and p.view.state.reinf_ms == -1 and m.view.state.reinf_ms == 0, "exact_single_value")
	_check(Private._packet_error(m) == "", "structural_validation")
	_check(_exact([p.view.state.tick, p.view.state.clock_ms, p.view.sim.sub_ms, p.view.state.over,
		p.view.state.reinf, p.view.state.reinf_ms, p.view.state.form_counter, p.view.state.next_id, p.view.state.next_event_id],
		[0, 0, 0, false, false, -1, 2, 3, 1]), "initial_values")
	_check(_exact(p.view.fleets.map(func(f): return f.id), [1, 2]) and
		_exact(p.view.fleets.map(func(f): return f.side), [0, 1]), "initial_roster")
	_check(p.view.missiles.is_empty() and p.view.swarms.is_empty(), "initial_projectiles")
	_check(_exact(p.view.sim._events, events) and _exact(p.view.sim._queue, [command])
		and p.view.sim.command_log.is_empty(), "initial_events_pending_log")
	names.append(NAMES[0])

	var b_sim := b._sim
	var b_state := b_sim.st
	var b_first := b_state.fleets[0]
	var b_second := b_state.fleets[1]
	_check(_exact(b.restore_completed(p), {"ok": true}), "positive_restore")
	_check(not is_same(b._sim, b_sim) and not is_same(b._sim.st, b_state)
		and not is_same(b._sim.st.fleets[0], b_first) and not is_same(b._sim.st.fleets[1], b_second), "positive_new_objects")
	_check(is_same(b._sim.st.by_id(1), b._sim.st.fleets[0])
		and is_same(b._sim.st.by_id(2), b._sim.st.fleets[1]), "positive_relink")
	_check(_exact(b.capture_completed(), p_result), "positive_exact")
	names.append(NAMES[1])

	var a_sim := a._sim
	var a_state := a_sim.st
	var first := a_state.by_id(1)
	var second := a_state.by_id(2)
	var cache: Dictionary = a_state._idx
	var before := a.capture_completed()
	var input_copy := m.duplicate(true)
	_check(_exact(before, p_result) and a._ready and not a._busy, "active_setup")
	_check(_exact(cache.keys(), [1, 2]) and a_state._idx_n == 2
		and is_same(cache[1], first) and is_same(cache[2], second), "active_warm_cache")
	_check(a_sim.salvo == null and a_sim.detect == null and a_sim.supply == null, "no_active_service")
	_check(_exact(a.restore_completed(m), {"ok": false, "error": "unsupported_reinforcement"}), "exact_rejection")
	_check(is_same(a._sim, a_sim) and is_same(a._sim.st, a_state)
		and is_same(a._sim.st.fleets[0], first) and is_same(a._sim.st.fleets[1], second), "active_identity")
	_check(is_same(a_state._idx, cache) and _exact(cache.keys(), [1, 2]) and a_state._idx_n == 2
		and is_same(cache[1], first) and is_same(cache[2], second), "active_cache_preserved")
	_check(a._ready and not a._busy and _exact(a.capture_completed(), before), "active_exact")
	_check(_exact(m, input_copy) and _exact(a_sim._queue, [command]) and _exact(a_sim._events, events), "input_pending_events")

	var applied := command.duplicate(true)
	applied.tick = 0
	events.append(_event("say", 1, Vector2(500, 500), "stop"))
	var after: Array[Dictionary] = []
	for owner in owners:
		_check(_exact(owner.advance(), {"ok": true}), "advance_" + str(after.size()))
		var next := owner.capture_completed()
		if not _check(next.ok, "capture_after_" + str(after.size())):
			_finish()
			return
		var view: Dictionary = next.packet.view
		_check(_exact([view.state.tick, view.state.clock_ms, view.sim.sub_ms, view.state.over, view.state.reinf,
			view.state.reinf_ms], [1, 100, 50, false, false, -1]), "time_oracle_" + str(after.size()))
		_check(_exact(view.fleets.map(func(f): return f.id), [1, 2]) and view.sim._queue.is_empty()
			and _exact(view.sim.command_log, [applied]) and _exact(view.sim._events, events), "suffix_oracle_" + str(after.size()))
		after.append(next)
	_check(_exact(after[0], after[2]) and _exact(after[1], after[2]), "full_post_advance_exact")
	names.append(NAMES[2])
	for owner in owners:
		owner.close()
	_finish()
